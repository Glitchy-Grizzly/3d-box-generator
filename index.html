function generateSCAD(w, l, h, wall, cx, cy, boxHex, textHex, text, font, gap) { currentScadCode = ` // Print-in-Place Hinged Box with Snap-Fit Latch // Fixed Hinge Alignment & Inlaid Text Exposure width = ${w}; length = ${l}; height = ${h}; wall = ${wall}; compX = ${cx}; compY = ${cy}; gap = ${gap}; tol = 0.3; $fn = 60; box_color = "${boxHex}"; text_color = "${textHex}"; box_text = "${text}"; font_style = "${font}:style=Bold"; module main() { color(box_color) { translate([0, -length/2 - gap/2, 0]) base_box(); translate([0, length/2 + gap/2, 0]) lid_box(); build_hinges(); } // Inlay Text (Sits inside the 0.6mm lid floor socket) if (box_text != "") { color(text_color) translate([0, length/2 + gap/2, 0]) linear_extrude(0.6) text(box_text, font=font_style, size=min(width/5, length/(len(box_text)*0.8)), halign="center", valign="center"); } } module base_box() { union() { difference() { translate([-width/2, -length/2, 0]) cube([width, length, height]); comp_w = (width - wall * (compX + 1)) / compX; comp_l = (length - wall * (compY + 1)) / compY; for(i=[0:compX-1]) { for(j=[0:compY-1]) { translate([ -width/2 + wall + i*(comp_w+wall), -length/2 + wall + j*(comp_l+wall), wall ]) cube([comp_w, comp_l, height]); } } } // Ramp Catch Wedge translate([-width/8, -length/2 - wall, height - wall*2]) rotate([0, 90, 0]) linear_extrude(width/4) polygon([[0,0], [wall*1.5, wall*1.5], [0, wall*1.5]]); } } module lid_box() { armW = width / 4; armT = wall; difference() { union() { translate([-width/2, -length/2, 0]) cube([width, length, height]); // Cantilever Snap Latch Structure translate([0, length/2, 0]) { translate([-armW/2, 0, 0]) cube([armW, armT, height]); translate([-armW/2, 0, height]) cube([wall, armT, wall*2]); translate([armW/2 - wall, 0, height]) cube([wall, armT, wall*2]); translate([-armW/2, 0, height + wall*2]) cube([armW, armT, wall]); } } // Hollow lid interior translate([-width/2 + wall, -length/2 + wall, wall]) cube([width - wall*2, length - wall*2, height + 1]); // Cut out clean socket on bottom face for text inlay if (box_text != "") { translate([0, 0, -0.05]) linear_extrude(0.7) text(box_text, font=font_style, size=min(width/5, length/(len(box_text)*0.8)), halign="center", valign="center"); } } } module build_hinges() { module teardrop(r, l) { rotate([0, 90, 0]) linear_extrude(l, center=true) polygon([ [0, -r], [r, 0], [0, r], [-r*0.7, r*0.7], [-r, 0], [-r*0.7, -r*0.7] ]); } knuckle_w = width / 5; hr = wall; pin_y_base = -gap/2 - wall/2; pin_y_lid = gap/2 + wall/2; pin_z = height; // BASE HINGE KNUCKLES for(i = [-2, 0, 2]) { translate([i * knuckle_w, pin_y_base, pin_z]) { teardrop(hr - tol, knuckle_w - tol); // Solid anchor block extending into Base box rear wall translate([-(knuckle_w - tol)/2, -wall/2, -hr]) cube([knuckle_w - tol, wall, hr]); } } // LID HINGE KNUCKLES for(i = [-1, 1]) { translate([i * knuckle_w, pin_y_lid, pin_z]) { difference() { union() { rotate([0, 90, 0]) cylinder(r=hr, h=knuckle_w - tol, center=true); // Solid anchor block extending into Lid box rear wall translate([-(knuckle_w - tol)/2, -wall/2, -hr]) cube([knuckle_w - tol, wall, hr]); } teardrop(hr + tol/2, knuckle_w + 1); } } } } main(); `; }
